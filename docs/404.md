<script>
  // The old Sandcastle API pages (api/1.0/<year>/...) moved to the owl site, which redirects html/*.htm itself.
  (function () {
    var match = location.pathname.match(/\/api\/1\.0\/(\d{4})\/(.*)$/i);
    if (!match) return;
    var year = Math.max({{ site.rir_api_oldest }}, parseInt(match[1], 10));
    var page = /^html\/[^\/]+\.htm$/i.test(match[2]) ? match[2] : '';
    location.replace('{{ site.rir_api_url }}/' + year + '/' + page + location.hash);
  })();
</script>
